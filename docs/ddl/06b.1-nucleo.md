# 06b.1 nucleo

DDL del esquema `nucleo`: locales, usuarios, roles, asignaciones por local, serie del tipo de cambio y contador de correlativos. Referencia de revisión del modelo de [06.1 nucleo](../modelo-datos/06.1-nucleo.md).

> En producción lo genera Django vía migraciones (`Meta.db_table` con nombre calificado por esquema). El orden de este archivo replica el orden de aplicación de las migraciones.

```sql
CREATE SCHEMA IF NOT EXISTS nucleo;

CREATE TABLE nucleo.local (
    id               BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre           VARCHAR(100) NOT NULL,
    codigo           VARCHAR(10)  NOT NULL,
    direccion        VARCHAR(200) NOT NULL,
    telefono         VARCHAR(30),
    persona_contacto VARCHAR(100),
    is_active        BOOLEAN NOT NULL DEFAULT true,
    CONSTRAINT local_nombre_uk UNIQUE (nombre),
    CONSTRAINT local_codigo_uk UNIQUE (codigo)
);

CREATE TABLE nucleo.usuario (
    id                     BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre_usuario         VARCHAR(50)  NOT NULL,
    correo                 VARCHAR(150) NOT NULL,
    hash_contrasena        VARCHAR(255) NOT NULL,
    is_global              BOOLEAN NOT NULL DEFAULT false,
    preferencias           JSONB NOT NULL DEFAULT '{}',
    is_active              BOOLEAN NOT NULL DEFAULT true,
    local_predeterminado_id BIGINT NULL REFERENCES nucleo.local(id),
    creado_en              TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en         TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT usuario_nombre_usuario_uk UNIQUE (nombre_usuario),
    CONSTRAINT usuario_correo_uk UNIQUE (correo)
);

CREATE TABLE nucleo.usuario_local (
    id         BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    usuario_id BIGINT NOT NULL REFERENCES nucleo.usuario(id),
    local_id   BIGINT NOT NULL REFERENCES nucleo.local(id),
    CONSTRAINT usuario_local_usuario_local_uk UNIQUE (usuario_id, local_id)
);

CREATE TABLE nucleo.rol (
    id     BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    codigo VARCHAR(50)  NOT NULL,
    nombre VARCHAR(100) NOT NULL,
    is_active BOOLEAN NOT NULL DEFAULT true,
    CONSTRAINT rol_uk UNIQUE (codigo)
);

CREATE TABLE nucleo.usuario_rol (
    id         BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    usuario_id BIGINT NOT NULL REFERENCES nucleo.usuario(id),
    rol_id     BIGINT NOT NULL REFERENCES nucleo.rol(id),
    local_id   BIGINT NULL REFERENCES nucleo.local(id),  -- NULL = rige globalmente
    CONSTRAINT usuario_rol_uk UNIQUE (usuario_id, rol_id, local_id)
);

-- Serie temporal del tipo de cambio: vigente = ultimo por registrada_en DESC
CREATE TABLE nucleo.tasa_cambio (
    id              BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    registrada_en   TIMESTAMPTZ NOT NULL,
    tasa_usd_a_bs   NUMERIC(14, 6) NOT NULL CHECK (tasa_usd_a_bs > 0),
    registrada_por_id BIGINT NULL REFERENCES nucleo.usuario(id)
);

-- Numeración correlativa por (tipo_documento, local, anio), con bloqueo de fila
CREATE TABLE nucleo.contador_correlativo (
    id             BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    tipo_documento VARCHAR(30) NOT NULL,
    local_id       BIGINT NOT NULL REFERENCES nucleo.local(id),
    anio           SMALLINT NOT NULL,
    ultimo         INTEGER NOT NULL DEFAULT 0 CHECK (ultimo >= 0),
    CONSTRAINT contador_correlativo_uk UNIQUE (tipo_documento, local_id, anio)
);

CREATE INDEX idx_tasa_cambio_vigencia ON nucleo.tasa_cambio (registrada_en DESC);
```
