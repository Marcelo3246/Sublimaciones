# 06b.3 catalogo

DDL del esquema `catalogo`: categorías, subcategorías, unidades de medida, tallas, materiales, productos, variantes y servicios. Referencia de revisión del modelo de [06.3 catalogo](../modelo-datos/06.3-catalogo.md).

```sql
CREATE SCHEMA IF NOT EXISTS catalogo;

CREATE TABLE IF NOT EXISTS catalogo.categoria (
    id             BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nombre         VARCHAR(100) NOT NULL,
    tipo           VARCHAR(20)  NOT NULL DEFAULT 'MATERIAL',
    activo         BOOLEAN      NOT NULL DEFAULT true,
    creado_en      TIMESTAMPTZ  NOT NULL DEFAULT now(),
    actualizado_en TIMESTAMPTZ  NOT NULL DEFAULT now(),
    CONSTRAINT catalog_categoria_uk UNIQUE (nombre, tipo),
    CONSTRAINT catalog_categoria_tipo_ck CHECK (tipo IN ('MATERIAL', 'PRODUCTO', 'SERVICIO'))
);

CREATE TABLE IF NOT EXISTS catalogo.subcategoria (
    id             BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    categoria_id   BIGINT       NOT NULL REFERENCES catalogo.categoria(id) ON DELETE RESTRICT,
    nombre         VARCHAR(100) NOT NULL,
    activo         BOOLEAN      NOT NULL DEFAULT true,
    creado_en      TIMESTAMPTZ  NOT NULL DEFAULT now(),
    actualizado_en TIMESTAMPTZ  NOT NULL DEFAULT now(),
    CONSTRAINT catalog_subcategoria_uk UNIQUE (categoria_id, nombre)
);

CREATE TABLE IF NOT EXISTS catalogo.unidad_medida (
    id                BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    codigo            VARCHAR(10) NOT NULL,
    nombre            VARCHAR(50) NOT NULL,
    permite_decimales BOOLEAN     NOT NULL DEFAULT true,
    activo            BOOLEAN     NOT NULL DEFAULT true,
    creado_en         TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en    TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT catalog_unidad_medida_codigo_uk UNIQUE (codigo)
);

CREATE TABLE IF NOT EXISTS catalogo.talla (
    id             BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    codigo         VARCHAR(10) NOT NULL,
    nombre         VARCHAR(50) NOT NULL,
    orden          SMALLINT    NOT NULL DEFAULT 0 CHECK (orden >= 0),
    activo         BOOLEAN     NOT NULL DEFAULT true,
    creado_en      TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT catalog_talla_codigo_uk UNIQUE (codigo)
);

CREATE TABLE IF NOT EXISTS catalogo.material (
    id                  BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    codigo              VARCHAR(50)    NOT NULL,
    nombre              VARCHAR(150)   NOT NULL,
    categoria_id        BIGINT         NOT NULL REFERENCES catalogo.categoria(id) ON DELETE RESTRICT,
    unidad_medida_id    BIGINT         NOT NULL REFERENCES catalogo.unidad_medida(id) ON DELETE RESTRICT,
    es_continuo         BOOLEAN        NOT NULL DEFAULT false,
    stock_minimo_global NUMERIC(12, 4) NULL,
    activo              BOOLEAN        NOT NULL DEFAULT true,
    creado_en           TIMESTAMPTZ    NOT NULL DEFAULT now(),
    actualizado_en      TIMESTAMPTZ    NOT NULL DEFAULT now(),
    CONSTRAINT catalog_material_codigo_uk UNIQUE (codigo)
);

CREATE TABLE IF NOT EXISTS catalogo.producto (
    id              BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    codigo          VARCHAR(50)  NOT NULL,
    nombre          VARCHAR(150) NOT NULL,
    descripcion     TEXT         NOT NULL DEFAULT '',
    subcategoria_id BIGINT       NULL REFERENCES catalogo.subcategoria(id) ON DELETE RESTRICT,
    tipo            VARCHAR(20)  NOT NULL DEFAULT 'TERMINADO',
    activo          BOOLEAN      NOT NULL DEFAULT true,
    creado_en       TIMESTAMPTZ  NOT NULL DEFAULT now(),
    actualizado_en  TIMESTAMPTZ  NOT NULL DEFAULT now(),
    CONSTRAINT catalog_producto_codigo_uk UNIQUE (codigo),
    CONSTRAINT catalog_producto_tipo_ck CHECK (tipo IN ('TERMINADO', 'SEMITERMINADO', 'SERVICIO'))
);

-- Todo producto stockeable lleva al menos una variante (talla_id NULL = única)
CREATE TABLE IF NOT EXISTS catalogo.variante_producto (
    id              BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    producto_id     BIGINT      NOT NULL REFERENCES catalogo.producto(id) ON DELETE RESTRICT,
    talla_id        BIGINT      NULL REFERENCES catalogo.talla(id) ON DELETE RESTRICT,
    color           VARCHAR(50) NOT NULL DEFAULT '',
    sku             VARCHAR(30) NOT NULL,
    codigo_barras   VARCHAR(50) NOT NULL DEFAULT '',
    activo          BOOLEAN     NOT NULL DEFAULT true,
    creado_en       TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en  TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT catalog_variante_producto_sku_uk UNIQUE (sku),
    CONSTRAINT catalog_variante_uk UNIQUE (producto_id, talla_id)
);

CREATE TABLE IF NOT EXISTS catalogo.servicio (
    id                BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nombre            VARCHAR(150) NOT NULL,
    descripcion       TEXT         NOT NULL DEFAULT '',
    unidad_medida_id  BIGINT       NULL REFERENCES catalogo.unidad_medida(id) ON DELETE RESTRICT,
    activo            BOOLEAN      NOT NULL DEFAULT true,
    creado_en         TIMESTAMPTZ  NOT NULL DEFAULT now(),
    actualizado_en    TIMESTAMPTZ  NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS catalog_material_categoria_idx
    ON catalogo.material (categoria_id, activo);

CREATE INDEX IF NOT EXISTS catalog_producto_subcat_idx
    ON catalogo.producto (subcategoria_id, activo);
```
