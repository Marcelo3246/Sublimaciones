# 06b.3 catalogo

DDL del esquema `catalogo`: categorías, subcategorías, unidades de medida, tallas, materiales, productos, variantes y servicios. Referencia de revisión del modelo de [06.3 catalogo](../modelo-datos/06.3-catalogo.md).

```sql
CREATE SCHEMA IF NOT EXISTS catalogo;

CREATE TABLE catalogo.categoria (
    id     BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    tipo   VARCHAR(20) NOT NULL CHECK (tipo IN ('material','producto','servicio')),
    is_active BOOLEAN NOT NULL DEFAULT true,
    creado_en      TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT categoria_uk UNIQUE (nombre, tipo)
);

CREATE TABLE catalogo.subcategoria (
    id           BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    categoria_id BIGINT NOT NULL REFERENCES catalogo.categoria(id),
    nombre       VARCHAR(100) NOT NULL,
    is_active    BOOLEAN NOT NULL DEFAULT true,
    CONSTRAINT subcategoria_uk UNIQUE (categoria_id, nombre)
);

CREATE TABLE catalogo.unidad_medida (
    id     BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    codigo VARCHAR(10) NOT NULL,             -- 'm', 'm2', 'l', 'g', 'u'
    nombre VARCHAR(50) NOT NULL,
    tipo   VARCHAR(20) NOT NULL CHECK (tipo IN ('longitud','area','volumen','masa','unidad')),
    CONSTRAINT unidad_medida_uk UNIQUE (codigo)
);

CREATE TABLE catalogo.talla (
    id     BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    codigo VARCHAR(10) NOT NULL,             -- 'S', 'M', 'L', 'XL', '38'
    nombre VARCHAR(50) NOT NULL,
    tipo   VARCHAR(20) NOT NULL CHECK (tipo IN ('alfabetica','numerica')),
    orden  SMALLINT NOT NULL,
    CONSTRAINT talla_uk UNIQUE (codigo)
);

CREATE TABLE catalogo.material (
    id              BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre          VARCHAR(150) NOT NULL,
    categoria_id    BIGINT NOT NULL REFERENCES catalogo.categoria(id),
    subcategoria_id BIGINT NULL REFERENCES catalogo.subcategoria(id),
    formato_stock   VARCHAR(10) NOT NULL CHECK (formato_stock IN ('continuo','contable')),
    tipo_medicion   VARCHAR(20) NOT NULL CHECK (tipo_medicion IN ('longitud','area','volumen','masa','unidad')),
    um_base_id      BIGINT NOT NULL REFERENCES catalogo.unidad_medida(id),
    sobrante_minimo NUMERIC(12, 4),           -- en unidades de um_base; NULL = sin minimo
    is_active       BOOLEAN NOT NULL DEFAULT true,
    creado_en       TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en  TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT material_um_sobrante_check CHECK (
        formato_stock = 'continuo' OR (sobrante_minimo IS NULL)
    )
);

CREATE TABLE catalogo.producto (
    id              BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre          VARCHAR(150) NOT NULL,
    categoria_id    BIGINT NOT NULL REFERENCES catalogo.categoria(id),
    subcategoria_id BIGINT NULL REFERENCES catalogo.subcategoria(id),
    material_id     BIGINT NULL REFERENCES catalogo.material(id),
    is_active       BOOLEAN NOT NULL DEFAULT true,
    creado_en       TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en  TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- Todo producto stockeable lleva al menos una variante (talla_id NULL = unica)
CREATE TABLE catalogo.variante_producto (
    id                      BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    producto_id             BIGINT NOT NULL REFERENCES catalogo.producto(id),
    talla_id                BIGINT NULL REFERENCES catalogo.talla(id),
    sku                     VARCHAR(30) NOT NULL,
    largo_estandar_material NUMERIC(12, 4),
    is_active               BOOLEAN NOT NULL DEFAULT true,
    creado_en               TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en          TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT variante_producto_sku_uk UNIQUE (sku),
    CONSTRAINT variante_producto_uk UNIQUE (producto_id, talla_id) -- PostgreSQL: NULLs distintos
);

CREATE TABLE catalogo.servicio (
    id              BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre          VARCHAR(150) NOT NULL,
    categoria_id    BIGINT NOT NULL REFERENCES catalogo.categoria(id),
    subcategoria_id BIGINT NULL REFERENCES catalogo.subcategoria(id),
    um_base_id      BIGINT NULL REFERENCES catalogo.unidad_medida(id),
    is_active       BOOLEAN NOT NULL DEFAULT true,
    creado_en       TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en  TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_variante_producto_activo ON catalogo.variante_producto (producto_id) WHERE is_active;
```
