# 06b.7 ventas

DDL del esquema `ventas`: clientes, pedidos, líneas, cobros y facturas propias. Referencia de revisión del modelo de [06.7 ventas](../modelo-datos/06.7-ventas.md).

```sql
CREATE SCHEMA IF NOT EXISTS ventas;

CREATE TABLE ventas.cliente (
    id        BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre    VARCHAR(150) NOT NULL,
    telefono  VARCHAR(30),
    correo    VARCHAR(150),
    rif       VARCHAR(20),                   -- NULL = consumidor final
    is_active BOOLEAN NOT NULL DEFAULT true,
    creado_en TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_cliente_nombre ON ventas.cliente (nombre);

CREATE TABLE ventas.pedido_venta (
    id                       BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    codigo                   VARCHAR(30) NOT NULL,
    cliente_id               BIGINT NOT NULL REFERENCES ventas.cliente(id),  -- obligatorio
    local_id                 BIGINT NOT NULL REFERENCES nucleo.local(id),
    estado                   VARCHAR(20) NOT NULL DEFAULT 'cotizado'
                             CHECK (estado IN ('cotizado','confirmado','en_produccion','listo','entregado','facturado','anulado')),
    moneda                   VARCHAR(3) NOT NULL CHECK (moneda IN ('USD','BS')),
    total                    NUMERIC(14, 2) NOT NULL DEFAULT 0 CHECK (total >= 0),
    vendedor_trabajador_id   BIGINT,         -- referencia debil a rrhh.trabajador
    confirmado_en            TIMESTAMPTZ,
    fecha_compromiso         DATE,
    motivo_anulacion         VARCHAR(200),   -- obligatorio al anular (RF-43)
    is_active                BOOLEAN NOT NULL DEFAULT true,
    creado_en                TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en           TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT pedido_codigo_uk UNIQUE (codigo),
    CONSTRAINT pedido_anulacion_motivo CHECK (
        estado <> 'anulado' OR motivo_anulacion IS NOT NULL
    )
);
CREATE INDEX idx_pedido_estado_local ON ventas.pedido_venta (estado, local_id);
CREATE INDEX idx_pedido_cliente ON ventas.pedido_venta (cliente_id, creado_en DESC);

CREATE TABLE ventas.linea_pedido (
    id             BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    pedido_venta_id BIGINT NOT NULL REFERENCES ventas.pedido_venta(id),
    tipo_linea     VARCHAR(30) NOT NULL CHECK (tipo_linea IN ('producto','producto_personalizado','servicio_puro')),
    variante_id    BIGINT NULL REFERENCES catalogo.variante_producto(id),
    servicio_id    BIGINT NULL REFERENCES catalogo.servicio(id),
    material_id    BIGINT NULL REFERENCES catalogo.material(id),
    um_id          BIGINT NULL REFERENCES catalogo.unidad_medida(id),
    medida_cantidad NUMERIC(12, 4),
    um_medida_id   BIGINT NULL REFERENCES catalogo.unidad_medida(id),
    cantidad       NUMERIC(12, 4) NOT NULL CHECK (cantidad > 0),
    precio_unitario NUMERIC(14, 2) NOT NULL CHECK (precio_unitario >= 0),
    moneda         VARCHAR(3) NOT NULL CHECK (moneda IN ('USD','BS')),
    item_precio_id BIGINT,                  -- referencia debil a precios.item_matriz_precios
    nota           VARCHAR(300),
    CONSTRAINT linea_producto_variante CHECK (
        tipo_linea NOT IN ('producto','producto_personalizado') OR variante_id IS NOT NULL
    ),
    CONSTRAINT linea_servicio CHECK (
        tipo_linea <> 'servicio_puro' OR servicio_id IS NOT NULL
    ),
    CONSTRAINT linea_personalizado_material CHECK (
        tipo_linea <> 'producto_personalizado' OR material_id IS NOT NULL
    ),
    CONSTRAINT linea_medida_um CHECK (
        (medida_cantidad IS NULL AND um_medida_id IS NULL)
        OR (medida_cantidad IS NOT NULL AND um_medida_id IS NOT NULL)
    )
);
CREATE INDEX idx_linea_pedido ON ventas.linea_pedido (pedido_venta_id);

CREATE TABLE ventas.cobro (
    id               BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    pedido_venta_id  BIGINT NOT NULL REFERENCES ventas.pedido_venta(id),
    monto            NUMERIC(14, 2) NOT NULL CHECK (monto > 0),
    moneda           VARCHAR(3) NOT NULL CHECK (moneda IN ('USD','BS')),
    metodo           VARCHAR(20) NOT NULL CHECK (metodo IN ('efectivo','punto','transferencia','divisa')),
    referencia       VARCHAR(100),           -- numero de operacion bancaria
    tasa_aplicada_id BIGINT NULL REFERENCES nucleo.tasa_cambio(id),
    fecha            TIMESTAMPTZ NOT NULL DEFAULT now(),
    anulado_en       TIMESTAMPTZ NULL,              -- registra cuándo fue anulado; NULL = cobro vigente
    is_active        BOOLEAN NOT NULL DEFAULT true,  -- false = cobro anulado (borrado lógico)
    creado_en        TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_cobro_pedido ON ventas.cobro (pedido_venta_id);

CREATE TABLE ventas.factura (
    id               BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    pedido_venta_id  BIGINT NOT NULL REFERENCES ventas.pedido_venta(id),
    local_id         BIGINT NOT NULL REFERENCES nucleo.local(id),
    numero           VARCHAR(20) NOT NULL,
    anio             SMALLINT NOT NULL,
    tipo             VARCHAR(10) NOT NULL DEFAULT 'propia' CHECK (tipo IN ('propia','fiscal')),
    estado           VARCHAR(10) NOT NULL DEFAULT 'emitida' CHECK (estado IN ('emitida','anulada')),
    total            NUMERIC(14, 2) NOT NULL CHECK (total >= 0),
    moneda           VARCHAR(3) NOT NULL CHECK (moneda IN ('USD','BS')),
    fecha_emision    DATE NOT NULL DEFAULT CURRENT_DATE,
    motivo_anulacion VARCHAR(200),
    emitida_por_id   BIGINT NOT NULL REFERENCES nucleo.usuario(id),
    is_active        BOOLEAN NOT NULL DEFAULT true,
    creado_en        TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en   TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT factura_correlativo_uk UNIQUE (tipo, local_id, anio, numero),
    CONSTRAINT factura_anulacion CHECK (
        estado <> 'anulada' OR motivo_anulacion IS NOT NULL
    )
);
CREATE INDEX idx_factura_repo ON ventas.factura (tipo, local_id, fecha_emision);
```
