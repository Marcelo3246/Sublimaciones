# 06b.6 produccion

DDL del esquema `produccion`: órdenes, pasos, consumos, máquinas y operaciones. Referencia de revisión del modelo de [06.6 produccion](../modelo-datos/06.6-produccion.md).

```sql
CREATE SCHEMA IF NOT EXISTS produccion;

CREATE TABLE produccion.orden_produccion (
    id               BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    codigo           VARCHAR(30) NOT NULL,     -- correlativo por (tipo, local, anio)
    pedido_venta_id  BIGINT,                   -- referencia debil a ventas.pedido_venta
    variante_id      BIGINT NOT NULL REFERENCES catalogo.variante_producto(id),
    cantidad         NUMERIC(12, 4) NOT NULL CHECK (cantidad > 0),
    local_origen_id  BIGINT NOT NULL REFERENCES nucleo.local(id),
    estado           VARCHAR(20) NOT NULL DEFAULT 'creada'
                     CHECK (estado IN ('creada','en_produccion','terminada','anulada')),
    fecha_limite     DATE,
    costo_materiales NUMERIC(14, 2),           -- se llena al cierre (RF-35)
    costo_horas      NUMERIC(14, 2),
    costo_total      NUMERIC(14, 2),
    moneda           VARCHAR(3) NOT NULL DEFAULT 'USD' CHECK (moneda IN ('USD','BS')),
    fecha_cierre     TIMESTAMPTZ,
    is_active        BOOLEAN NOT NULL DEFAULT true,
    creado_en        TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en   TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT orden_produccion_codigo_uk UNIQUE (codigo),
    CONSTRAINT orden_produccion_cierre_costo CHECK (
        estado <> 'terminada' OR (costo_total IS NOT NULL AND fecha_cierre IS NOT NULL)
    )
);
CREATE INDEX idx_orden_produccion_estado ON produccion.orden_produccion (estado, local_origen_id);
CREATE INDEX idx_orden_produccion_pedido ON produccion.orden_produccion (pedido_venta_id);

CREATE TABLE produccion.paso_produccion (
    id                   BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    orden_produccion_id  BIGINT NOT NULL REFERENCES produccion.orden_produccion(id),
    orden_paso           SMALLINT NOT NULL CHECK (orden_paso > 0),
    tipo_paso            VARCHAR(20) NOT NULL CHECK (tipo_paso IN ('impresion','corte','planchado','confeccion','diseno')),
    local_id             BIGINT NOT NULL REFERENCES nucleo.local(id),
    trabajador_id        BIGINT NULL REFERENCES rrhh.trabajador(id),
    estado               VARCHAR(20) NOT NULL DEFAULT 'pendiente'
                         CHECK (estado IN ('pendiente','en_proceso','terminado')),
    iniciado_en          TIMESTAMPTZ,
    terminado_en         TIMESTAMPTZ,
    CONSTRAINT paso_orden_item_uk UNIQUE (orden_produccion_id, orden_paso),
    CONSTRAINT paso_fechas_check CHECK (iniciado_en IS NULL OR terminado_en IS NULL OR iniciado_en <= terminado_en)
);
CREATE INDEX idx_paso_local_estado ON produccion.paso_produccion (local_id, estado);

CREATE TABLE produccion.consumo_material (
    id                 BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    paso_id            BIGINT NOT NULL REFERENCES produccion.paso_produccion(id),
    lote_id            BIGINT NULL REFERENCES inventario.lote(id),
    stock_id           BIGINT NULL REFERENCES inventario.stock(id),
    cantidad_consumida NUMERIC(12, 4) NOT NULL CHECK (cantidad_consumida > 0),
    um_id              BIGINT NOT NULL REFERENCES catalogo.unidad_medida(id),
    tipo_consumo       VARCHAR(20) NOT NULL CHECK (tipo_consumo IN ('uso','desperdicio')),
    motivo_desperdicio VARCHAR(200),
    movimiento_stock_id BIGINT,             -- referencia debil a inventario.movimiento_stock
    CONSTRAINT consumo_excluyente CHECK (
        (lote_id IS NOT NULL AND stock_id IS NULL)
        OR (lote_id IS NULL AND stock_id IS NOT NULL)
    ),
    CONSTRAINT consumo_desperdicio_motivo CHECK (
        tipo_consumo <> 'desperdicio' OR motivo_desperdicio IS NOT NULL
    )
);
CREATE INDEX idx_consumo_paso ON produccion.consumo_material (paso_id);

CREATE TABLE produccion.maquina (
    id          BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre      VARCHAR(100) NOT NULL,
    local_id    BIGINT NOT NULL REFERENCES nucleo.local(id),
    tipo_maquina VARCHAR(20) NOT NULL CHECK (tipo_maquina IN ('impresora','plancha','cortadora','otro')),
    estado      VARCHAR(20) NOT NULL DEFAULT 'operativa'
                CHECK (estado IN ('operativa','mantenimiento','inactiva')),
    is_active   BOOLEAN NOT NULL DEFAULT true,
    creado_en   TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_maquina_local ON produccion.maquina (local_id);

CREATE TABLE produccion.operacion_maquina (
    id                    BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    paso_id               BIGINT NOT NULL REFERENCES produccion.paso_produccion(id),
    maquina_id            BIGINT NOT NULL REFERENCES produccion.maquina(id),
    operador_trabajador_id BIGINT NOT NULL REFERENCES rrhh.trabajador(id),
    duracion_minutos      NUMERIC(8, 2) NOT NULL CHECK (duracion_minutos >= 0),
    resultado             VARCHAR(20) NOT NULL CHECK (resultado IN ('completado','defectuoso','parcial')),
    nota                  VARCHAR(300),
    creado_en             TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_operacion_maquina ON produccion.operacion_maquina (maquina_id, creado_en DESC);
```

> Nota de orden: `produccion.paso_produccion` referencia `rrhh.trabajador`, que se crea después. La migración de Django aplica `rrhh` antes que `produccion`.
