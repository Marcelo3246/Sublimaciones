# 06b.10 documentos

DDL del esquema `documentos`: plantillas editables y repositorio de documentos emitidos. Referencia de revisión del modelo de [06.10 documentos](../modelo-datos/06.10-documentos.md).

```sql
CREATE SCHEMA IF NOT EXISTS documentos;

CREATE TABLE documentos.plantilla_documento (
    id               BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    tipo_documento   VARCHAR(30) NOT NULL CHECK (tipo_documento IN (
        'factura','nota_entrega','orden_produccion','planilla_transferencia',
        'acta_desperdicio','comprobante_comision')),
    nombre           VARCHAR(150) NOT NULL,
    ruta_archivo     VARCHAR(500),         -- imagen de fondo (logos, datos fiscales)
    contenido        TEXT,                 -- HTML/DSL de la plantilla
    is_predeterminada BOOLEAN NOT NULL DEFAULT false,
    is_active        BOOLEAN NOT NULL DEFAULT true,
    creado_en        TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en   TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE documentos.documento (
    id                 BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    tipo_documento     VARCHAR(30) NOT NULL CHECK (tipo_documento IN (
        'factura','nota_entrega','orden_produccion','planilla_transferencia',
        'acta_desperdicio','comprobante_comision')),
    local_id           BIGINT NOT NULL REFERENCES nucleo.local(id),
    numero             VARCHAR(20) NOT NULL,
    anio               SMALLINT NOT NULL,
    entidad_tipo       VARCHAR(50),       -- tipo del módulo de origen (referencia débil, ej: 'pedido_venta')
    entidad_id         BIGINT,
    plantilla_id       BIGINT NULL REFERENCES documentos.plantilla_documento(id),
    estado             VARCHAR(10) NOT NULL DEFAULT 'generado' CHECK (estado IN ('generado','anulado')),
    ruta_archivo       VARCHAR(500) NOT NULL,   -- PDF emitido
    generado_por_id    BIGINT NOT NULL REFERENCES nucleo.usuario(id),
    creado_en          TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT documento_correlativo_uk UNIQUE (tipo_documento, local_id, anio, numero),
    CONSTRAINT documento_ref_requerida CHECK (
        (entidad_tipo IS NOT NULL AND entidad_id IS NOT NULL)
        OR (entidad_tipo IS NULL AND entidad_id IS NULL)
    )
);
-- Los tres tipos de documento de venta se numeran por nucleo.contador_correlativo;
-- el PDF emitido queda en ruta_archivo, nunca se guarda en FileField sin backup.
CREATE INDEX idx_documento_repo ON documentos.documento (tipo_documento, local_id, numero);
```
