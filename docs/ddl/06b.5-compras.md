# 06b.5 compras

DDL del esquema `compras`: empresas, proveedores, órdenes y líneas, recepciones, facturas e historial de precios. Referencia de revisión del modelo de [06.5 compras](../modelo-datos/06.5-compras.md).

```sql
CREATE SCHEMA IF NOT EXISTS compras;

CREATE TABLE compras.empresa (
    id        BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre    VARCHAR(150) NOT NULL,
    rif       VARCHAR(20) NOT NULL,
    direccion VARCHAR(200),
    telefono  VARCHAR(30),
    correo    VARCHAR(150),
    is_active BOOLEAN NOT NULL DEFAULT true,
    creado_en TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT empresa_rif_uk UNIQUE (rif)
);

CREATE TABLE compras.proveedor (
    id                     BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    empresa_id             BIGINT NOT NULL REFERENCES compras.empresa(id),
    persona_contacto       VARCHAR(100) NOT NULL,
    telefono               VARCHAR(30),
    correo                 VARCHAR(150),
    moneda_predeterminada  VARCHAR(3) NOT NULL DEFAULT 'USD' CHECK (moneda_predeterminada IN ('USD','BS')),
    condiciones_pago       VARCHAR(200),
    is_active              BOOLEAN NOT NULL DEFAULT true,
    creado_en              TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en         TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_proveedor_empresa ON compras.proveedor (empresa_id);

CREATE TABLE compras.orden_compra (
    id            BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    proveedor_id  BIGINT NOT NULL REFERENCES compras.proveedor(id),
    local_id      BIGINT NOT NULL REFERENCES nucleo.local(id),
    estado        VARCHAR(20) NOT NULL DEFAULT 'borrador'
                  CHECK (estado IN ('borrador','enviada','parcial','recibida','cerrada','anulada')),
    total         NUMERIC(14, 2) NOT NULL DEFAULT 0 CHECK (total >= 0),
    moneda        VARCHAR(3) NOT NULL CHECK (moneda IN ('USD','BS')),
    fecha_emision DATE NOT NULL DEFAULT CURRENT_DATE,
    fecha_envio   DATE,
    is_active     BOOLEAN NOT NULL DEFAULT true,
    creado_en     TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_orden_compra_estado ON compras.orden_compra (local_id, estado);

CREATE TABLE compras.linea_orden_compra (
    id                   BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    orden_compra_id      BIGINT NOT NULL REFERENCES compras.orden_compra(id),
    material_id          BIGINT NOT NULL REFERENCES catalogo.material(id),
    um_id                BIGINT NOT NULL REFERENCES catalogo.unidad_medida(id),
    cantidad_solicitada  NUMERIC(12, 4) NOT NULL CHECK (cantidad_solicitada > 0),
    cantidad_recibida    NUMERIC(12, 4) NOT NULL DEFAULT 0 CHECK (cantidad_recibida >= 0),
    precio_unitario      NUMERIC(14, 2) NOT NULL CHECK (precio_unitario >= 0),
    moneda               VARCHAR(3) NOT NULL CHECK (moneda IN ('USD','BS')),
    CONSTRAINT linea_orden_cantidad_check CHECK (cantidad_recibida <= cantidad_solicitada)
);
CREATE INDEX idx_linea_orden_compra ON compras.linea_orden_compra (orden_compra_id);

CREATE TABLE compras.recepcion_compra (
    id               BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    orden_compra_id  BIGINT NOT NULL REFERENCES compras.orden_compra(id),
    local_id         BIGINT NOT NULL REFERENCES nucleo.local(id),
    estado           VARCHAR(20) NOT NULL DEFAULT 'pendiente'
                     CHECK (estado IN ('pendiente','total','parcial','anulada')),
    fecha_recepcion  DATE NOT NULL DEFAULT CURRENT_DATE,
    observacion      VARCHAR(300),
    is_active        BOOLEAN NOT NULL DEFAULT true,
    creado_en        TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en   TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_recepcion_compra ON compras.recepcion_compra (orden_compra_id);

CREATE TABLE compras.linea_recepcion_compra (
    id                     BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    recepcion_compra_id    BIGINT NOT NULL REFERENCES compras.recepcion_compra(id),
    linea_orden_compra_id  BIGINT NOT NULL REFERENCES compras.linea_orden_compra(id),
    cantidad_recibida      NUMERIC(12, 4) NOT NULL CHECK (cantidad_recibida > 0),
    um_id                  BIGINT NOT NULL REFERENCES catalogo.unidad_medida(id),
    lote_id                BIGINT NULL REFERENCES inventario.lote(id)  -- lote generado solo si el material es continuo
);
CREATE INDEX idx_linea_recepcion ON compras.linea_recepcion_compra (recepcion_compra_id);

CREATE TABLE compras.factura_compra (
    id                    BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    proveedor_id          BIGINT NOT NULL REFERENCES compras.proveedor(id),
    orden_compra_id       BIGINT NULL REFERENCES compras.orden_compra(id),
    recepcion_compra_id   BIGINT NULL REFERENCES compras.recepcion_compra(id),
    numero_externo        VARCHAR(50),           -- numero impreso por el proveedor
    fecha_emision         DATE NOT NULL DEFAULT CURRENT_DATE,
    total                 NUMERIC(14, 2) NOT NULL CHECK (total >= 0),
    moneda                VARCHAR(3) NOT NULL CHECK (moneda IN ('USD','BS')),
    ruta_archivo          VARCHAR(500),          -- PDF/imagen adjunta
    is_active             BOOLEAN NOT NULL DEFAULT true,
    creado_en             TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en        TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_factura_compra_proveedor ON compras.factura_compra (proveedor_id, fecha_emision);
CREATE INDEX idx_factura_compra_moneda_fecha ON compras.factura_compra (moneda, fecha_emision);

CREATE TABLE compras.historial_precio_proveedor (
    id              BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    proveedor_id    BIGINT NOT NULL REFERENCES compras.proveedor(id),
    empresa_id      BIGINT NOT NULL REFERENCES compras.empresa(id),   -- heredado para agrupar
    material_id     BIGINT NOT NULL REFERENCES catalogo.material(id),
    um_id           BIGINT NOT NULL REFERENCES catalogo.unidad_medida(id),
    precio_unitario NUMERIC(14, 2) NOT NULL CHECK (precio_unitario >= 0),
    moneda          VARCHAR(3) NOT NULL CHECK (moneda IN ('USD','BS')),
    registrado_en   TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_historial_precio ON compras.historial_precio_proveedor
    (material_id, proveedor_id, registrado_en DESC);
```
