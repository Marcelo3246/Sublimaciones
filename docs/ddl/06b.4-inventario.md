# 06b.4 inventario

DDL del esquema `inventario`: saldos por local, lotes, movimientos, guías de transferencia y niveles mínimos. Referencia de revisión del modelo de [06.4 inventario](../modelo-datos/06.4-inventario.md).

```sql
CREATE SCHEMA IF NOT EXISTS inventario;

CREATE TABLE inventario.stock (
    id                   BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    tipo                 VARCHAR(20) NOT NULL CHECK (tipo IN ('material','producto')),
    material_id          BIGINT NULL REFERENCES catalogo.material(id),
    variante_id          BIGINT NULL REFERENCES catalogo.variante_producto(id),
    local_id             BIGINT NOT NULL REFERENCES nucleo.local(id),
    um_id                BIGINT NOT NULL REFERENCES catalogo.unidad_medida(id),
    cantidad_disponible  NUMERIC(12, 4) NOT NULL DEFAULT 0 CHECK (cantidad_disponible >= 0),
    cantidad_reservada   NUMERIC(12, 4) NOT NULL DEFAULT 0 CHECK (cantidad_reservada >= 0),
    estado               VARCHAR(20) NOT NULL DEFAULT 'disponible'
                         CHECK (estado IN ('disponible','danado','en_transito')),
    is_active            BOOLEAN NOT NULL DEFAULT true,
    creado_en            TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en       TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT stock_excluyente CHECK (
        (material_id IS NOT NULL AND variante_id IS NULL)
        OR (material_id IS NULL AND variante_id IS NOT NULL)
    ),
    CONSTRAINT stock_unica_fila UNIQUE (local_id, material_id, variante_id) -- ver nota de NULLs
);
-- Nota: en PostgreSQL 14, UNIQUE trata cada NULL como distinto; para garantizar
-- una sola fila por clave se usa un indice unico funcional:
CREATE UNIQUE INDEX idx_stock_unica_fila ON inventario.stock
  (local_id, tipo, COALESCE(material_id, 0), COALESCE(variante_id, 0)) WHERE is_active;

CREATE TABLE inventario.lote (
    id               BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    material_id      BIGINT NOT NULL REFERENCES catalogo.material(id),
    local_id         BIGINT NOT NULL REFERENCES nucleo.local(id),
    codigo_lote      VARCHAR(30) NOT NULL,
    um_id            BIGINT NOT NULL REFERENCES catalogo.unidad_medida(id),
    cantidad_inicial NUMERIC(12, 4) NOT NULL CHECK (cantidad_inicial > 0),
    cantidad_actual  NUMERIC(12, 4) NOT NULL CHECK (cantidad_actual >= 0),
    fecha_vencimiento DATE NULL,
    costo_unitario   NUMERIC(14, 2) NOT NULL CHECK (costo_unitario >= 0),
    moneda           VARCHAR(3) NOT NULL CHECK (moneda IN ('USD','BS')),
    estado           VARCHAR(20) NOT NULL DEFAULT 'disponible'
                     CHECK (estado IN ('disponible','reservado','en_transito','danado','vencido','descartado')),
    motivo_descarte  VARCHAR(200),
    is_active        BOOLEAN NOT NULL DEFAULT true,
    creado_en        TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en   TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT lote_codigo_uk UNIQUE (codigo_lote),
    CONSTRAINT lote_descarte_motivo CHECK (
        estado NOT IN ('descartado') OR motivo_descarte IS NOT NULL
    )
);
CREATE INDEX idx_lote_local_material ON inventario.lote (local_id, material_id, estado);

-- Movimientos de stock: unico camino que toca stock y lote
CREATE TABLE inventario.movimiento_stock (
    id               BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    tipo_movimiento  VARCHAR(30) NOT NULL CHECK (tipo_movimiento IN (
        'entrada_compra','entrada_produccion','entrada_transferencia','entrada_devolucion',
        'entrada_directa','salida_venta','salida_consumo','salida_desperdicio',
        'salida_transferencia','ajuste_entrada','ajuste_salida','reserva','liberacion_reserva')),
    local_id         BIGINT NOT NULL REFERENCES nucleo.local(id),
    local_destino_id BIGINT NULL REFERENCES nucleo.local(id),
    lote_id          BIGINT NULL REFERENCES inventario.lote(id),
    stock_id         BIGINT NULL REFERENCES inventario.stock(id),
    cantidad         NUMERIC(12, 4) NOT NULL CHECK (cantidad > 0),
    um_id            BIGINT NOT NULL REFERENCES catalogo.unidad_medida(id),
    usuario_id       BIGINT NOT NULL REFERENCES nucleo.usuario(id),
    documento_origen_tipo VARCHAR(40),     -- tipo del documento que originó el movimiento (ej: 'pedido_venta', 'orden_produccion')
    documento_origen_id   BIGINT,          -- id del documento origen (referencia debil)
    motivo           VARCHAR(200),
    creado_en        TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT movimiento_referencia_unica CHECK (
        (lote_id IS NOT NULL AND stock_id IS NULL)
        OR (lote_id IS NULL AND stock_id IS NOT NULL)
    ),
    CONSTRAINT movimiento_traslado_destino CHECK (
        tipo_movimiento NOT IN ('salida_transferencia','entrada_transferencia')
        OR (local_destino_id IS NOT NULL)
    )
);
CREATE INDEX idx_movimiento_local_fecha ON inventario.movimiento_stock (local_id, creado_en DESC);
CREATE INDEX idx_movimiento_ref ON inventario.movimiento_stock (documento_origen_tipo, documento_origen_id);

CREATE TABLE inventario.guia_transferencia (
    id               BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    codigo_guia      VARCHAR(30) NOT NULL,
    local_origen_id  BIGINT NOT NULL REFERENCES nucleo.local(id),
    local_destino_id BIGINT NOT NULL REFERENCES nucleo.local(id),
    estado           VARCHAR(20) NOT NULL DEFAULT 'solicitada'
                     CHECK (estado IN ('solicitada','aprobada','despachada','en_transito','recibida','cancelada')),
    fecha_solicitud  TIMESTAMPTZ NOT NULL DEFAULT now(),
    fecha_despacho   TIMESTAMPTZ,
    fecha_recepcion  TIMESTAMPTZ,
    creada_por_id    BIGINT NOT NULL REFERENCES nucleo.usuario(id),
    observacion      VARCHAR(300),
    is_active        BOOLEAN NOT NULL DEFAULT true,
    creado_en        TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en   TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT guia_codigo_uk UNIQUE (codigo_guia)
);

CREATE TABLE inventario.linea_guia_transferencia (
    id          BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    guia_id     BIGINT NOT NULL REFERENCES inventario.guia_transferencia(id),
    lote_id     BIGINT NULL REFERENCES inventario.lote(id),
    variante_id BIGINT NULL REFERENCES catalogo.variante_producto(id),
    cantidad    NUMERIC(12, 4) NOT NULL CHECK (cantidad > 0),
    um_id       BIGINT NOT NULL REFERENCES catalogo.unidad_medida(id),
    estado      VARCHAR(20) NOT NULL DEFAULT 'pendiente'
                CHECK (estado IN ('pendiente','despachada','recibida','desviada')),
    CONSTRAINT linea_guia_excluyente CHECK (
        (lote_id IS NOT NULL AND variante_id IS NULL)
        OR (lote_id IS NULL AND variante_id IS NOT NULL)
    )
);
CREATE INDEX idx_linea_guia ON inventario.linea_guia_transferencia (guia_id);

CREATE TABLE inventario.nivel_minimo (
    id          BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    material_id BIGINT NOT NULL REFERENCES catalogo.material(id),
    local_id    BIGINT NOT NULL REFERENCES nucleo.local(id),
    um_id       BIGINT NOT NULL REFERENCES catalogo.unidad_medida(id),
    cantidad    NUMERIC(12, 4) NOT NULL CHECK (cantidad >= 0),
    is_active   BOOLEAN NOT NULL DEFAULT true,
    creado_en   TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT nivel_minimo_uk UNIQUE (material_id, local_id)
);
```
