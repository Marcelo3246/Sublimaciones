# 06b.8 precios

DDL del esquema `precios`: matriz de precios, historial de cambios y parámetros de la fórmula. Referencia de revisión del modelo de [06.8 precios](../modelo-datos/06.8-precios.md).

```sql
CREATE SCHEMA IF NOT EXISTS precios;

CREATE TABLE precios.item_matriz_precios (
    id                BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    producto_id       BIGINT NULL REFERENCES catalogo.producto(id),
    variante_id       BIGINT NULL REFERENCES catalogo.variante_producto(id),
    servicio_id       BIGINT NULL REFERENCES catalogo.servicio(id),
    material_id       BIGINT NULL REFERENCES catalogo.material(id),
    um_id             BIGINT NULL REFERENCES catalogo.unidad_medida(id),
    medida_cantidad   NUMERIC(12, 4),
    um_medida_id      BIGINT NULL REFERENCES catalogo.unidad_medida(id),
    precio_base       NUMERIC(14, 2) NOT NULL CHECK (precio_base >= 0),
    moneda            VARCHAR(3) NOT NULL CHECK (moneda IN ('USD','BS')),
    porcentaje_margen NUMERIC(5, 4) NOT NULL CHECK (porcentaje_margen > 0),
    porcentaje_iva    NUMERIC(5, 4) NOT NULL CHECK (porcentaje_iva >= 0),
    is_active         BOOLEAN NOT NULL DEFAULT true,
    creado_en         TIMESTAMPTZ NOT NULL DEFAULT now(),
    actualizado_en    TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT item_matriz_destino_excluyente CHECK (
        (producto_id IS NOT NULL AND servicio_id IS NULL)
        OR (producto_id IS NULL AND variante_id IS NOT NULL AND servicio_id IS NULL)
        OR (producto_id IS NULL AND variante_id IS NULL AND servicio_id IS NOT NULL)
    )
);
CREATE INDEX idx_item_matriz_producto ON precios.item_matriz_precios (producto_id);
CREATE INDEX idx_item_matriz_variante ON precios.item_matriz_precios (variante_id);
CREATE INDEX idx_item_matriz_servicio ON precios.item_matriz_precios (servicio_id);

CREATE TABLE precios.historial_cambio_precio (
    id              BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    item_matriz_id  BIGINT NOT NULL REFERENCES precios.item_matriz_precios(id),
    precio_anterior NUMERIC(14, 2) NOT NULL,
    precio_nuevo    NUMERIC(14, 2) NOT NULL,
    usuario_id      BIGINT NOT NULL REFERENCES nucleo.usuario(id),
    cambiado_en     TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_historial_cambio ON precios.historial_cambio_precio (item_matriz_id, cambiado_en DESC);

-- Fórmula parametrizable por vigencia (RF-45)
CREATE TABLE precios.parametro_formula (
    id                BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    porcentaje_margen NUMERIC(5, 4) NOT NULL CHECK (porcentaje_margen > 0),
    porcentaje_iva    NUMERIC(5, 4) NOT NULL CHECK (porcentaje_iva >= 0),
    fecha_desde       DATE NOT NULL,
    fecha_hasta       DATE NULL,             -- NULL = vigente hasta que exista el siguiente
    registrado_por_id BIGINT NOT NULL REFERENCES nucleo.usuario(id),
    CONSTRAINT parametro_formula_uk UNIQUE (fecha_desde)
);
```
